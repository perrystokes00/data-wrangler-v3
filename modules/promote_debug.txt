    SELECT DISTINCT TRY_CONVERT(DATETIME2, NULLIF(LTRIM(RTRIM([ABANDONMENT_DATE])), '')) AS [abandonment_date], LEFT(CONVERT(CHAR(40), HASHBYTES('SHA1', CAST(UPPER(LTRIM(RTRIM([ASSIGNED_FIELD]))) AS NVARCHAR(4000))), 2), 40) AS [assigned_field], TRY_CONVERT(DATETIME2, NULLIF(LTRIM(RTRIM([COMPLETION_DATE])), '')) AS [completion_date], [DEEPEST_DEPTH] AS [deepest_depth], LEFT([ELEV_REF_DATUM], 40) AS [elev_ref_datum], [KB_ELEV] AS [kb_elev], LEFT([LEASE_NAME], 255) AS [lease_name], LEFT([LOCATION_TYPE], 40) AS [location_type], LEFT(CONVERT(CHAR(40), HASHBYTES('SHA1', CAST(UPPER(LTRIM(RTRIM([CURR_OPERATOR]))) AS NVARCHAR(4000))), 2), 40) AS [operator], TRY_CONVERT(DATETIME2, NULLIF(LTRIM(RTRIM([SPUD_DATE])), '')) AS [spud_date], [SURFACE_LATITUDE] AS [surface_latitude], [SURFACE_LONGITUDE] AS [surface_longitude], LEFT([UWI], 40) AS [uwi], LEFT([WELL_NAME], 255) AS [well_name]
    FROM [stg].[stg_kgs_wells_load_all] src
    WHERE src.[UWI] IS NOT NULL AND LTRIM(RTRIM(src.[UWI])) <> ''
) dedup
WHERE NOT EXISTS (SELECT 1 FROM [dbo].[well] tgt WHERE tgt.[UWI] = dedup.[UWI])
